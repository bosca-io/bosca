@Generated
public class TestRepositoryImpl : TestRepository {
  private val json: Json by lazy { runBlocking { get<Json>() } }

  private val mappers: DefaultMappers by lazy { DefaultMappers(json) }

  @OptIn(ExperimentalCoroutinesApi::class)
  override suspend fun purgeExpired(): List<UUID> = withContext(DatabaseDispatcher) {
     val connection = connection()
     connection.useStatement("DELETE FROM tokens WHERE expires < now() RETURNING principal_id") { stmt ->
         connection.markNeedsCommitOrRollback()
                      
         stmt.executeQuery().use { row ->
             val results = mutableListOf<UUID>()
             while (row.next()) {
                 results.add(
     mappers.uuid.map(UUID::class, emptyList<KClass<*>>(), row, 1) ?: error("row value is null")
                 )
             }
             return@useStatement results
         }
     }
  }
}
